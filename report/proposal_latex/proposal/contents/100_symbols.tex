\begin{longtable}{p{3.2cm}cp{10.3cm}}

$\mathbf{w}$ & : & Vektor bobot (proporsi) portofolio utang \\
$w_i$ & : & Bobot (proporsi) instrumen utang ke-$i$ \\
$n$ & : & Jumlah instrumen dalam portofolio \\
$T$ & : & Jumlah periode pengamatan \\
$C_{i,t}$ & : & Nilai biaya instrumen ke-$i$ pada periode ke-$t$ \\
$\Delta c_{i,t}$ & : & Perubahan biaya logaritmik instrumen ke-$i$ pada periode ke-$t$ \\
$\overline{\Delta c}_i$ & : & Rata-rata perubahan biaya instrumen ke-$i$ \\
$\boldsymbol{\mu}$ & : & Vektor biaya ekspektasi instrumen \\
$\sigma_i^2$ & : & Variansi perubahan biaya instrumen ke-$i$ \\
$\sigma_p^2$ & : & Variansi perubahan biaya portofolio \\
$\boldsymbol{\Sigma}$ & : & Matriks kovarians perubahan biaya \\
$\mathrm{Cov}(\Delta c_i,\Delta c_j)$ & : & Kovarians perubahan biaya instrumen ke-$i$ dan ke-$j$ \\
$FV_i$ & : & Nilai instrumen ke-$i$ setelah penyesuaian nilai tukar dan suku bunga \\
$FX_i$ & : & Nilai tukar instrumen ke-$i$ terhadap USD \\
$IR_i$ & : & Tingkat suku bunga (\textit{yield}) instrumen ke-$i$ \\
$S_t$ & : & Nilai \textit{market rate} pada waktu ke-$t$ (model GBM) \\
$\mu$ & : & Parameter \textit{drift} pada model GBM \\
$\sigma$ & : & Parameter volatilitas pada model GBM \\
$W_t$ & : & Proses Wiener (gerak Brown) \\
$Z$ & : & Bilangan acak berdistribusi normal baku \\
$\mathrm{Cost}_s(\mathbf{w})$ & : & Total biaya portofolio pada skenario ke-$s$ \\
$\alpha$ & : & Tingkat kepercayaan pada CCaR \\
$\mathcal{S}_\alpha$ & : & Himpunan skenario dengan biaya tertinggi \\
$\mathrm{CCaR}_\alpha(\mathbf{w})$ & : & \textit{Conditional Cost-at-Risk} portofolio pada tingkat kepercayaan $\alpha$ \\
$\mathrm{MCCaR}_{p,i}$ & : & \textit{Marginal Conditional Cost-at-Risk} instrumen ke-$i$ \\
$\mathrm{MC}_i$ & : & Biaya marginal instrumen ke-$i$ \\
$\tau$ & : & Batas maksimum risiko ekor portofolio \\
$lb_i,\,ub_i$ & : & Batas bawah dan batas atas proporsi instrumen ke-$i$ \\
$k_i$ & : & Posisi individu (solusi) ke-$i$ dalam KMA \\
$k_{\text{best}}$ & : & Solusi terbaik global pada KMA \\
$f(k_i)$ & : & Nilai \textit{fitness} individu ke-$i$ \\
$q$ & : & Jumlah individu \textit{big males} \\
$d$ & : & \textit{Mlipir rate} \\
$\varepsilon$ & : & Toleransi konvergensi \\
$k_{\max}$ & : & Jumlah iterasi maksimum \\

\end{longtable}
